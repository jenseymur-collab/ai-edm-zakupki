# 10.3 IMPLEMENTATION (CORE MODULE)  
  
```python  
import re  
from typing import Dict, List  
  
FORBIDDEN_PATTERNS = [  
    r"should exist",  
    r"assume",  
    r"pseudo",  
    r"magic happens",  
    r"TODO",  
]  
  
ARCHITECTURE_RED_FLAGS = [  
    r"direct DB",  
    r"SELECT .* FROM",  
    r"bypass gateway",  
]  
  
def detect_issues(text: str) -> List[str]:  
    issues = []  
  
    for p in FORBIDDEN_PATTERNS:  
        if re.search(p, text, re.IGNORECASE):  
            issues.append("hallucination_pattern")  
  
    for p in ARCHITECTURE_RED_FLAGS:  
        if re.search(p, text, re.IGNORECASE):  
            issues.append("architecture_violation")  
  
    return issues  
  
  
def sanitize_code(code: str) -> str:  
    code = re.sub(r"should exist", "must be implemented explicitly", code)  
    code = re.sub(r"assume.*", "", code)  
    code = re.sub(r"\.\.\.", "# TODO: implement properly", code)  
    return code  
  
  
def rewrite_safe_architecture(code: str) -> str:  
    return """  
# AUTO-REWRITTEN SAFE VERSION  
  
class ServiceClient:  
    def __init__(self, api_client):  
        self.api_client = api_client  
  
    def execute(self, payload):  
        return self.api_client.call_service(payload)  
  
# Original logic removed due to architecture risk  
"""  
  
  
def fix_response(response: Dict, issues: List[str]) -> Dict:  
    code = response.get("code", "")  
    explanation = response.get("explanation", "")  
    next_steps = response.get("next_steps", "")  
  
    if not explanation:  
        explanation = "Safe implementation following standard patterns."  
  
    if not next_steps:  
        next_steps = "Run tests → validate → integrate"  
  
    if "architecture_violation" in issues:  
        code = rewrite_safe_architecture(code)  
  
    code = sanitize_code(code)  
  
    return {  
        "code": code,  
        "explanation": explanation,  
        "next_steps": next_steps,  
        "meta": {  
            "auto_fixed": True,  
            "issues_corrected": issues  
        }  
    }  
  
  
def process_llm_response(response: Dict) -> Dict:  
    text = response.get("code", "") + response.get("explanation", "")  
    issues = detect_issues(text)  
  
    if not issues:  
        return response  
  
    return fix_response(response, issues)
